        function verRespuestas() {
            const r1 = document.getElementById('ans1');
            const r2 = document.getElementById('ans2');
            const btn = document.querySelector('.btn-reveal');

            if (r1.style.display === "none" || r1.style.display === "") {
                r1.style.display = "block";
                r2.style.display = "block";
                btn.innerText = "Ocultar Respuestas";
            } else {
                r1.style.display = "none";
                r2.style.display = "none";
                btn.innerText = "Ver Respuestas";
            }
        }
